SELECT l1."GeometryID", l1."RestrictionTypeID", l2."GeometryID"
FROM import_geojson."Merged_Bays_LineString" l1 LEFT JOIN import_geojson."Merged_Bays_LineString_Singl" l2 ON l1."GeometryID" = l2."GeometryID"
WHERE l1."RestrictionTypeID" NOT IN (103, 147)
AND l2."GeometryID" IS NULL
ORDER BY l1."GeometryID"
